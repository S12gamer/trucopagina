#!/bin/bash

echo "Ingresa la ruta del directorio que deseas respaldar:"
read directorio
if [ ! -d "$directorio" ]; then
    echo "Error: El directorio no existe."
    exit 1
fi

nombre=$(basename "$directorio")
fecha=$(date +"%Y-%m-%d")

RESPALDO_DIR="$HOME/respaldos"
if [ ! -d "$RESPALDO_DIR" ]; then
    mkdir -p "$RESPALDO_DIR"
fi

mkdir -p "$RESPALDO_DIR/$nombre"
for archivo in $(find "$directorio" -type f); do
    cp "$archivo" "$RESPALDO_DIR/$nombre"
done
tar -czf "$RESPALDO_DIR/${nombre}_${fecha}.tar.gz" -C "$RESPALDO_DIR" "$nombre"

if [ $? -eq 0 ]; then
    echo "Respaldo creado correctamente."
else
    echo "Error al crear el respaldo."
    exit 1
fi
rm -rf "$RESPALDO_DIR/$nombre"
echo "Proceso finalizado."
